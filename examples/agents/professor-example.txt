Input: "How should we handle inventory updates when multiple users try to purchase the last item?"
Output:
**Problem Analysis**: Race condition in inventory management where concurrent purchases could oversell items.

**Proposed Solution**: Implement optimistic locking with PostgreSQL's row-level locks:
```typescript
async function purchaseItem(
  itemId: string,
  quantity: number,
  userId: string,
): Promise<PurchaseResult> {
  return await db.transaction(async (trx) => {
    // Lock the row for update
    const item = await trx("inventory")
      .where({ id: itemId })
      .forUpdate()
      .first();

    if (!item || item.quantity < quantity) {
      throw new InsufficientInventoryError(
        itemId,
        item?.quantity || 0,
        quantity,
      );
    }

    // Update inventory
    await trx("inventory")
      .where({ id: itemId })
      .update({ quantity: item.quantity - quantity });

    // Create order record
    const order = await trx("orders")
      .insert({
        user_id: userId,
        item_id: itemId,
        quantity,
        status: "pending",
      })
      .returning("*");
    return { success: true, orderId: order[0].id };
  });
}
```
